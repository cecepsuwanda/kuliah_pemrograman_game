\section{Bagian A: Pilihan Ganda}

\begin{enumerate}
  \item GameObject minimal memiliki komponen:
  \begin{enumerate}
    \item Rigidbody
    \item Transform
    \item Collider
    \item Script
  \end{enumerate}
  
  \item Method untuk gerakan fisika (Rigidbody) sebaiknya di:
  \begin{enumerate}
    \item Update
    \item FixedUpdate
    \item LateUpdate
    \item Awake
  \end{enumerate}
  
  \item OnTriggerEnter digunakan untuk:
  \begin{enumerate}
    \item Mendeteksi tabrakan fisik
    \item Mendeteksi overlap tanpa respon fisik
    \item Inisialisasi
    \item Render
  \end{enumerate}
  
  \item Prefab berguna untuk:
  \begin{enumerate}
    \item Hanya menyimpan material
    \item Template GameObject yang bisa diinstansiasi ulang
    \item Mengganti texture
    \item Build game
  \end{enumerate}
\end{enumerate}
