\begin{aktivitas}
  \item \textbf{Analisis Game}: Pilih satu game favorit dan analisis menggunakan MDA framework. Identifikasi mechanics, dynamics, dan aesthetics-nya.
  
  \item \textbf{Analisis Genre}: Pilih 3 game dari genre berbeda. Identifikasi genre utama dan karakteristik yang membuat game tersebut termasuk dalam genre tersebut.
  
  \item \textbf{Membuat GDD}: Buat Game Design Document untuk konsep game sederhana dengan Executive Summary, Game Overview, Core Gameplay Mechanics, dan Development Timeline.
  
  \item \textbf{Timeline Pengembangan}: Buat timeline pengembangan untuk game sederhana dengan breakdown fase pre-production, production, post-production beserta milestone.
\end{aktivitas}

\begin{asesmen}

\textbf{A. Pilihan Ganda}

\begin{enumerate}
  \item Manakah yang BUKAN merupakan elemen MDA framework?
  \begin{enumerate}
    \item Mechanics
    \item Dynamics
    \item Animation
    \item Aesthetics
  \end{enumerate}
  
  \item Fase dalam lifecycle pengembangan game yang fokus pada asset creation dan programming adalah:
  \begin{enumerate}
    \item Pre-production
    \item Production
    \item Post-production
    \item Prototyping
  \end{enumerate}
  
  \item Dokumen yang menjelaskan konsep, mekanik, dan desain game secara detail disebut:
  \begin{enumerate}
    \item Technical Design Document
    \item Game Design Document
    \item Project Plan
    \item Art Bible
  \end{enumerate}
\end{enumerate}

\begin{enumerate}

\begin{rangkuman}
Bab ini membahas landasan teori pengembangan game, termasuk definisi game, MDA framework, genre game, Game Design Document, dan lifecycle pengembangan.

\textbf{Poin Kunci:}
\begin{itemize}
  \item Game adalah aktivitas terstruktur dengan tujuan, aturan, dan umpan balik
  \item MDA framework: Mechanics (aturan), Dynamics (perilaku), Aesthetics (pengalaman)
  \item Genre game mengklasifikasikan game berdasarkan mekanik dan gaya permainan
  \item GDD adalah blueprint pengembangan game
  \item Lifecycle: pre-production, production, post-production
\end{itemize}
\end{rangkuman}
