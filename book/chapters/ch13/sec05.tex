\section{Distribution}

\begin{learningoutcomes}
\item Memahami distribution channels
\item Mengupload ke platform distribusi
\item Memahami requirements untuk setiap platform
\end{learningoutcomes}

\subsection{Distribution Channels}

\begin{itemize}
\item \textbf{itch.io}: Platform indie-friendly
\item \textbf{Steam}: PC distribution besar
\item \textbf{Google Play}: Android
\item \textbf{App Store}: iOS
\item \textbf{Web}: Host sendiri atau platform web
\end{itemize}

\subsection{itch.io Upload}

\begin{enumerate}
\item Buat akun itch.io
\item Create new project
\item Upload build files
\item Set metadata (screenshots, description)
\item Publish
\end{enumerate}

\subsection{Requirements}

\begin{itemize}
\item \textbf{Screenshots}: Minimal 1-5 screenshots
\item \textbf{Description}: Deskripsi game
\item \textbf{Trailer}: Video gameplay (optional)
\item \textbf{Icon}: Game icon
\end{itemize}

\begin{catatan}
Distribution memerlukan persiapan metadata dan assets marketing. Research requirements untuk setiap platform sebelum upload.
\end{catatan}

\subsection{Ringkasan Section}

\begin{itemize}
\item Various distribution channels available
\item itch.io untuk indie developers
\item Steam untuk PC distribution
\item Google Play dan App Store untuk mobile
\item Requirements berbeda untuk setiap platform
\end{itemize}
