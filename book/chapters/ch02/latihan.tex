\section{Latihan}

\begin{latihan}[Setup Project]
\begin{enumerate}
\item Install Unity Hub dan Unity Editor (versi LTS terbaru)
\item Buat project baru dengan template 3D
\item Eksplorasi struktur folder project
\item Buka Project Settings dan review konfigurasi default
\end{enumerate}
\end{latihan}

\begin{latihan}[Eksplorasi Interface]
\begin{enumerate}
\item Buka semua panel utama Unity Editor
\item Coba setiap tool di toolbar
\item Buat beberapa GameObject dan eksplorasi Hierarchy
\item Pilih GameObject dan lihat Inspector
\item Coba berbagai layout preset
\end{enumerate}
\end{latihan}

\begin{latihan}[Setup Workflow]
\begin{enumerate}
\item Buat struktur folder yang terorganisir dalam Project
\item Buat scene baru dan setup basic hierarchy
\item Buat beberapa GameObject dan convert ke Prefab
\item Test workflow dengan membuat simple scene
\end{enumerate}
\end{latihan}

\begin{latihan}[Eksplorasi Asset Store]
\begin{enumerate}
\item Buka Asset Store dan browse kategori
\item Cari asset gratis yang menarik
\item Install asset ke project
\item Eksplorasi Package Manager
\item Install package dari Unity Registry (misalnya: Cinemachine atau Post Processing)
\end{enumerate}
\end{latihan}
