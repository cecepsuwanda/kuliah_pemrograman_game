\section{Latihan}

\begin{latihan}[Membuat Skrip Pertama]
\begin{enumerate}
\item Buat skrip baru bernama "HelloWorld.cs"
\item Tulis pesan "Hello World" di fungsi Start()
\item Lampirkan ke GameObject dan jalankan game
\item Lihat output di Console
\end{enumerate}
\end{latihan}

\begin{latihan}[Eksperimen Lifecycle]
Buat skrip yang menampilkan log untuk setiap lifecycle function dan amati urutan eksekusinya.
\end{latihan}

\begin{latihan}[Menggunakan Variabel]
Buat skrip dengan berbagai tipe variabel dan coba manipulasi nilainya di Inspector dan melalui kode.
\end{latihan}

\begin{latihan}[Kontrol Alur]
Buat skrip yang menggunakan if-else, switch, dan loop untuk mengimplementasikan sistem kesehatan sederhana.
\end{latihan}

\begin{latihan}[OOP]
Buat hierarki class untuk berbagai jenis enemy dengan inheritance dan polymorphism.
\end{latihan}

\begin{latihan}[Coroutine]
Buat coroutine yang menampilkan countdown dan kemudian spawn enemy setiap 2 detik.
\end{latihan}

\begin{latihan}[Event System]
Implementasikan event system untuk komunikasi antara Player, Enemy, dan UI Manager.
\end{latihan}
