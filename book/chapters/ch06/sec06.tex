\section{Trigger untuk Interaksi}

\begin{learningoutcomes}
\item Menggunakan trigger untuk collectible
\item Mengimplementasikan interaction zones
\item Membuat checkpoint system
\end{learningoutcomes}

\subsection{Collectible System}

\begin{lstlisting}[language={[Sharp]C}]
public class Collectible : MonoBehaviour
{
    public int value = 10;
    
    void OnTriggerEnter(Collider other)
    {
        if (other.CompareTag("Player"))
        {
            Collect(other.GetComponent<Player>());
        }
    }
    
    void Collect(Player player)
    {
        player.AddScore(value);
        Destroy(gameObject);
    }
}
\end{lstlisting}

\subsection{Interaction Zone}

\begin{lstlisting}[language={[Sharp]C}]
public class InteractionZone : MonoBehaviour
{
    public string interactionText = "Press E to interact";
    private bool playerInRange = false;
    
    void OnTriggerEnter(Collider other)
    {
        if (other.CompareTag("Player"))
        {
            playerInRange = true;
            ShowInteractionUI();
        }
    }
    
    void OnTriggerExit(Collider other)
    {
        if (other.CompareTag("Player"))
        {
            playerInRange = false;
            HideInteractionUI();
        }
    }
    
    void Update()
    {
        if (playerInRange && Input.GetKeyDown(KeyCode.E))
        {
            Interact();
        }
    }
    
    void Interact()
    {
        // Interaction logic
    }
}
\end{lstlisting}

\begin{catatan}
Trigger sangat berguna untuk interaksi non-physical seperti collectible, checkpoint, atau interaction zones. Lebih efisien daripada collision untuk deteksi overlap.
\end{catatan}

\subsection{Ringkasan Section}

\begin{itemize}
\item Trigger digunakan untuk deteksi overlap tanpa collision fisik
\item OnTriggerEnter/Stay/Exit untuk handle trigger events
\item Berguna untuk collectible, checkpoint, dan interaction zones
\item Lebih efisien daripada collision untuk deteksi overlap
\end{itemize}
