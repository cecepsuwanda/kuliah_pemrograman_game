\chapter{Quick Reference C\# untuk Unity}

\section{Lifecycle Functions}

\begin{lstlisting}[language={[Sharp]C}]
void Awake()        // Called before Start, even if disabled
void OnEnable()    // Called when object enabled
void Start()        // Called before first frame
void Update()       // Called every frame
void FixedUpdate()  // Called for physics (default: 50Hz)
void LateUpdate()   // Called after all Update()
void OnDisable()    // Called when object disabled
void OnDestroy()    // Called when object destroyed
\end{lstlisting}

\section{Common Unity Types}

\begin{lstlisting}[language={[Sharp]C}]
Vector3 position;      // 3D position (x, y, z)
Vector2 position2D;    // 2D position (x, y)
Quaternion rotation;   // Rotation
Color color;           // RGBA color
GameObject obj;        // GameObject reference
Transform transform;   // Transform component
\end{lstlisting}

\section{Common Methods}

\begin{lstlisting}[language={[Sharp]C}]
// GameObject
Instantiate(prefab, position, rotation);
Destroy(gameObject);
Destroy(gameObject, delay);

// Transform
transform.position = new Vector3(x, y, z);
transform.rotation = Quaternion.Euler(x, y, z);
transform.localScale = Vector3.one;

// Component
GetComponent<T>();
GetComponentInChildren<T>();
GetComponents<T>();

// Time
Time.deltaTime;        // Time since last frame
Time.fixedDeltaTime;   // Fixed timestep
Time.timeScale;        // Time scale (0 = paused)

// Input
Input.GetKey(KeyCode.Space);
Input.GetKeyDown(KeyCode.Space);
Input.GetAxis("Horizontal");
Input.GetMouseButton(0);

// Physics
Physics.Raycast(origin, direction, out hit);
Physics.OverlapSphere(center, radius);
\end{lstlisting}

\section{Coroutine}

\begin{lstlisting}[language={[Sharp]C}]
IEnumerator MyCoroutine()
{
    yield return new WaitForSeconds(1f);
    yield return null;  // Wait one frame
    yield return new WaitForEndOfFrame();
}
StartCoroutine(MyCoroutine());
\end{lstlisting}

\section{Events}

\begin{lstlisting}[language={[Sharp]C}]
public event System.Action OnEvent;
OnEvent?.Invoke();

public event System.Action<int> OnValueChanged;
OnValueChanged?.Invoke(value);
\end{lstlisting}
